=== МОНИТОРИНГ СИСТЕМЫ (В РЕАЛЬНОМ ВРЕМЕНИ) ===

# Мониторинг свободного места (обновление каждые 5 сек)
watch -n 5 'df -h / | tail -1'

# Мониторинг оперативной памяти (обновление каждые 5 сек)
watch -n 5 'free -h'

# Мониторинг нагрузки на диски (IO)
iostat -dx 5

# Мониторинг процессов PostgreSQL (индексы, вакуум, анализ)
ps aux | grep -E "postgres.*(BIND|INDEX|vacuum|analyze)" | grep -v grep

=== МОНИТОРИНГ POSTGRESQL ===
# Просмотр активных запросов (длительность и текст запроса)
sudo -u postgres psql -c "SELECT pid, state, now() - query_start AS duration, query FROM pg_stat_activity WHERE state <> 'idle' ORDER BY duration DESC;"

# Проверка применения настроек памяти
sudo -u postgres psql -c "SHOW shared_buffers; SHOW maintenance_work_mem; SHOW work_mem;"

# Просмотр размера базы данных
sudo -u postgres psql -c "SELECT pg_size_pretty(pg_database_size('gis'));"

# Проверка логов на наличие критических ошибок
sudo tail -50 /var/log/postgresql/postgresql-15-main.log | grep -i "error\|fatal\|panic"
